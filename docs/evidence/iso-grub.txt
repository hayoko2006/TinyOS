### /boot/grub/config.cfg（超时与主题来源）
# TinyOS GRUB 引导配置
# 本文件覆盖 live-build 模板 /usr/share/live/build/bootloaders/grub-pc/config.cfg，
# 由 binary 阶段的 binary_grub_cfg 复制为 ISO 内 /boot/grub/config.cfg，
# 并被 /boot/grub/grub.cfg 第一行 `source /boot/grub/config.cfg` 加载。
# grub-pc（BIOS）与 grub-efi（UEFI）共用这一份配置。
#
# 相对模板新增：set timeout / set timeout_style（live-build 模板只有 set default=0，
# 会导致菜单无限等待、不自动引导），并启用 TinyOS 品牌主题。

set default=0
set timeout=10
set timeout_style=menu

if [ x$feature_default_font_path = xy ] ; then
    font=unicode
else
    font=$prefix/unicode.pf2
fi

# Copied from the netinst image
if loadfont $font ; then
    set gfxmode=800x600
    set gfxpayload=keep
    insmod efi_gop
    insmod efi_uga
    insmod video_bochs
    insmod video_cirrus
else
    set gfxmode=auto
    insmod all_video
fi

insmod gfxterm
insmod png

# TinyOS 品牌主题（随 includes.binary 注入）；缺失时回退 live-build 默认主题
if [ -e /boot/grub/themes/tinyos/theme.txt ]; then
    set theme=/boot/grub/themes/tinyos/theme.txt
else
    source /boot/grub/theme.cfg
fi

terminal_output gfxterm

insmod play
play 960 440 1 0 4 440 1
